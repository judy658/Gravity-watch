### 4. JavaScript Dosyası

```javascript
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');

canvas.width = 800;
canvas.height = 600;

const gravity = 0.15;
const jump = -4.5;

let bird = {
    x: 50,
    y: 150,
    width: 30,
    height: 20,
    velocity: 0
};

let pipes = [];
let score = 0;

function drawBird() {
    ctx.fillStyle = 'yellow';
    ctx.fillRect(bird.x, bird.y, bird.width, bird.height);
}

function updateBird() {
    bird.velocity += gravity;
    bird.y += bird.velocity;

    if (bird.y + bird.height > canvas.height || bird.y < 0) {
        resetGame();
    }
}

function drawPipes() {
    ctx.fillStyle = 'green';
    pipes.forEach(pipe => {
        ctx.fillRect(pipe.x, pipe.y, pipe.width, pipe.height);
        ctx.fillRect(pipe.x, pipe.y + pipe.gap, pipe.width, canvas.height - (pipe.y + pipe.gap));
    });
}

function updatePipes() {
    if (frame % 100 === 0) {
        pipes.push({
            x: canvas.width,
            y: Math.random() * (canvas.height - 200) + 50,
            width: 50,
            height: 200,
            gap: 150
        });
    }

    pipes.forEach(pipe => {
        pipe.x -= 2;
    });

    if (pipes.some(pipe => bird.x < pipe.x + pipe.width && bird.x + bird.width > pipe.x &&
        (bird.y < pipe.y || bird.y + bird.height > pipe.y + pipe.gap))) {
        resetGame();
    }

    pipes = pipes.filter(pipe => pipe.x > -pipe.width);
}

function drawScore() {
    ctx.fillStyle = 'white';
    ctx.font = '24px Arial';
    ctx.fillText('Score: ' + score, 10, 30);
}

let frame = 0;

function updateScore() {
    if (frame % 50 === 0) {
        score++;
    }
}

function resetGame() {
    bird.y = 150;
    bird.velocity = 0;
    pipes = [];
    score = 0;
}

document.addEventListener('keydown', () => {
    bird.velocity = jump;
});

function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    drawBird();
    updateBird();

    drawPipes();
    updatePipes();

    drawScore();
    updateScore();

    frame++;

    requestAnimationFrame(gameLoop);
}

gameLoop();
```
