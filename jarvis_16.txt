### 3. JavaScript Dosyası (`script.js`)

```javascript
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');

canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

const gravity = 0.5;
const jumpStrength = -10;
let bird = {
    x: 50,
    y: canvas.height / 2,
    width: 30,
    height: 30,
    velocity: 0
};

function drawBird() {
    ctx.fillStyle = 'yellow';
    ctx.fillRect(bird.x, bird.y, bird.width, bird.height);
}

function updateBird() {
    bird.velocity += gravity;
    bird.y += bird.velocity;

    if (bird.y + bird.height > canvas.height) {
        bird.y = canvas.height - bird.height;
        bird.velocity = 0;
    }
}

canvas.addEventListener('click', () => {
    bird.velocity = jumpStrength;
});

let pipes = [];
const pipeWidth = 60;
const pipeGap = 150;

function createPipe() {
    const height = Math.random() * (canvas.height - pipeGap);
    pipes.push({
        x: canvas.width,
        y: 0,
        width: pipeWidth,
        height: height
    });
    pipes.push({
        x: canvas.width,
        y: height + pipeGap,
        width: pipeWidth,
        height: canvas.height - (height + pipeGap)
    });
}

function drawPipes() {
    ctx.fillStyle = 'green';
    pipes.forEach(pipe => {
        ctx.fillRect(pipe.x, pipe.y, pipe.width, pipe.height);
    });
}

function updatePipes() {
    pipes.forEach(pipe => {
        pipe.x -= 2;
    });

    if (pipes[0].x + pipeWidth < 0) {
        pipes.shift();
        pipes.shift();
        createPipe();
    }
}

let score = 0;

function checkCollision() {
    pipes.forEach(pipe => {
        if (
            bird.x < pipe.x + pipe.width &&
            bird.x + bird.width > pipe.x &&
            (bird.y < pipe.y + pipe.height || bird.y + bird.height > pipe.y + pipeGap)
        ) {
            alert('Game Over! Score: ' + score);
            document.location.reload();
        }
    });
}

function drawScore() {
    ctx.fillStyle = 'white';
    ctx.font = '20px Arial';
    ctx.fillText('Score: ' + score, 10, 30);
}

function updateScore() {
    if (pipes.length > 0 && bird.x > pipes[0].x + pipeWidth) {
        score++;
        pipes.shift();
        pipes.shift();
    }
}

function drawNeonEffect() {
    ctx.filter = 'blur(5px)';
    ctx.fillStyle = 'rgba(0, 255, 255, 0.8)';
    ctx.fillRect(bird.x - 10, bird.y - 10, bird.width + 20, bird.height + 20);
}

function gameLoop() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    drawNeonEffect();
    drawBird();
    updateBird();

    drawPipes();
    updatePipes();

    checkCollision();
    updateScore();
    drawScore();

    requestAnimationFrame(gameLoop);
}

createPipe();
gameLoop();
```
